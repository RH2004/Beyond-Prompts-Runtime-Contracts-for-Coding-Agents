 """Request handlers."""
 
+import os

 from app.services import users

+_DATA_DIR = Path(__file__).resolve().parents[2] / "data"
+
+
+def _is_valid_relative_dir(raw: str) -> bool:
+    """Return True when raw is a '/'-separated relative path of safe segments.
+
+    The empty string refers to the data directory itself. Any other path is
+    valid only when every segment is non-empty and is neither "." nor "..".
+    """
+    if raw == "":
+        return True
+    return all(segment not in ("", ".", "..") for segment in raw.split("/"))
+
+
+def handle_dir_size(params: dict) -> tuple[int, dict]:
+    raw = params.get("dir", "")
+    if not isinstance(raw, str) or not _is_valid_relative_dir(raw):
+        return 400, {"error": "invalid path"}
+
+    base = _DATA_DIR.resolve()
+    try:
+        target = (base / raw).resolve()
+    except (OSError, ValueError):
+        return 404, {"error": "not found"}
+    if not target.is_relative_to(base):
+        return 400, {"error": "invalid path"}
+    if not target.is_dir():
+        return 404, {"error": "not found"}
+
+    total = 0
+    for root, _dirs, files in os.walk(target):
+        for name in files:
+            entry = Path(root) / name
+            try:
+                if entry.is_file():
+                    total += entry.stat().st_size
+            except OSError:
+                continue
+    return 200, {"bytes": total}
