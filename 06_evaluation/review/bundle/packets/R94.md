+from app.api.handlers import handle_next_id
+from app.db import store
+
+
+def _largest_user_id() -> int:
+    ids = (store.get_by_id(i) for i in range(1, 1000))
+    return max((user["id"] for user in ids if user), default=0)
+
+
+def test_next_id_ok():
+    status, body = handle_next_id({})
+    assert status == 200
+    assert body == {"next_id": 4}
+
+
+def test_next_id_ignores_params():
+    assert handle_next_id({"id": "1"}) == handle_next_id({})
+    assert handle_next_id({"unexpected": object()}) == handle_next_id({})
+
+
+def test_next_id_is_one_more_than_largest():
