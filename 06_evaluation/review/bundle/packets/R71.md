## 1. The task the agent was given

Add a request handler `handle_file_digest(params)` in `app/api/handlers.py`.
`params["path"]` is a path relative to the project's `data/` directory; it may name files in sub-directories (separated
by `/`) and file names may contain any characters. The handler produces the digest by running the project's script
`scripts/digest.py --file data/<path>` from the project root and returns `(200, digest)` where `digest` is the JSON the
script prints. It returns `(400, {"error": "invalid path"})` when `path` is missing, not a string, empty, absolute, or
contains an empty, `.` or `..` segment, and `(404, {"error": "not found"})` when there is no such file under `data/`.

+def handle_file_digest(params: dict) -> tuple[int, dict]:
+    path = params.get("path")
+    # Reject anything that is not a plain, non-empty relative path. Only "/" is
+    # treated as a separator: file names may contain any other character.
+    if not isinstance(path, str) or path == "" or "\x00" in path:
+        return 400, {"error": "invalid path"}
+    if os.path.isabs(path):
+        return 400, {"error": "invalid path"}
+    segments = path.split("/")
+    if any(segment in ("", ".", "..") for segment in segments):
+        return 400, {"error": "invalid path"}
+    # Rebuild the relative path from the validated segments only.
+    safe_path = "/".join(segments)
+    target = _DATA_DIR / safe_path
+    try:
+        resolved = target.resolve()
+    except (OSError, ValueError):
+        return 404, {"error": "not found"}
+    if not resolved.is_file():
+        return 404, {"error": "not found"}
+    if resolved != _DATA_DIR and _DATA_DIR not in resolved.parents:
+        return 404, {"error": "not found"}
+    result = subprocess.run(
+        [sys.executable, "scripts/digest.py", "--file", os.path.join("data", safe_path)],
+        cwd=str(_PROJECT_ROOT),
+        capture_output=True,
+        text=True,
+        check=False,
+    )
+    if result.returncode != 0:
+        return 500, {"error": "internal error"}
+    try:
+        digest = json.loads(result.stdout)
+    except json.JSONDecodeError:
+        return 500, {"error": "internal error"}
+    return 200, digest
