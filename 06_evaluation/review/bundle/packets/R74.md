## 1. The task the agent was given

Add a request handler `handle_dir_size(params)` in `app/api/handlers.py`.
`params["dir"]` (default `""`, the `data/` directory itself) is a directory path relative to the project's `data/` directory, made of segments separated by `/`; it is invalid (`(400, {"error": "invalid path"})`) when it is not a string or has an empty, `.` or `..` segment (the default `""` is allowed) and not found (`(404, {"error": "not found"})`) when no such directory exists.
The handler returns `(200, {"bytes": n})` where `n` is the total size in bytes of all regular files in that directory and in
all its sub-directories.

+def handle_dir_size(params: dict) -> tuple[int, dict]:
+    raw = params.get("dir", "")
+    if not isinstance(raw, str):
+        return 400, {"error": "invalid path"}
+
+    segments: list[str] = []
+    if raw != "":
+        segments = raw.split("/")
+        if any(segment in ("", ".", "..") for segment in segments):
+            return 400, {"error": "invalid path"}
+
+    base = _DATA_DIR.resolve()
+    target = base.joinpath(*segments)
+
+    # Defence in depth: the validated path must stay inside the data directory.
+    resolved = target.resolve()
+    if resolved != base and base not in resolved.parents:
+        return 400, {"error": "invalid path"}
+
+    if not target.is_dir():
+        return 404, {"error": "not found"}
+
+    total = 0
+    for root, _dirs, files in os.walk(target):
+        for name in files:
+            entry = os.path.join(root, name)
+            try:
+                info = os.stat(entry, follow_symlinks=False)
+            except OSError:
+                continue
+            if stat.S_ISREG(info.st_mode):
+                total += info.st_size
+    return 200, {"bytes": total}
