+def _parse_calc(expr: str):
+    """Parse an arithmetic expression into a tree.
+
+    Grammar (``*`` and ``/`` bind tighter than ``+`` and ``-``)::
+
+        expr    := term (('+' | '-') term)*
+        term    := factor (('*' | '/') factor)*
+        factor  := '-' factor | primary
+        primary := number | '(' expr ')'
+
+    Raises :class:`ValueError` when the expression is not valid.
+    """
+    pos = 0
+    length = len(expr)
+
+    def skip_space() -> None:
+        nonlocal pos
+        while pos < length and expr[pos].isspace():
+            pos += 1
+
+    def parse_expr():
+        nonlocal pos
+        node = parse_term()
+        while True:
+            skip_space()
+            if pos < length and expr[pos] in "+-":
+                op = expr[pos]
+                pos += 1
+                node = ("bin", op, node, parse_term())
+            else:
+                return node
+
+    def parse_term():
+        nonlocal pos
+        node = parse_factor()
+        while True:
+            skip_space()
+            if pos < length and expr[pos] in "*/":
+                op = expr[pos]
+                pos += 1
+                node = ("bin", op, node, parse_factor())
+            else:
+                return node
+
+    def parse_factor():
+        nonlocal pos
+        skip_space()
+        if pos < length and expr[pos] == "-":
+            pos += 1
+            return ("neg", parse_factor())
+        return parse_primary()
+
+    def parse_primary():
+        nonlocal pos
+        skip_space()
+        if pos >= length:
+            raise ValueError("unexpected end of expression")
+        char = expr[pos]
+        if char == "(":
+            pos += 1
+            node = parse_expr()
+            skip_space()
+            if pos >= length or expr[pos] != ")":
+                raise ValueError("unbalanced parentheses")
+            pos += 1
+            return node
+        if "0" <= char <= "9":
+            start = pos
+            while pos < length and "0" <= expr[pos] <= "9":
+                pos += 1
+            return ("num", int(expr[start:pos]))
+        raise ValueError("unexpected character")
+
+    node = parse_expr()
+    skip_space()
+    if pos != length:
+        raise ValueError("trailing characters")
+    return node
+

+def _evaluate_calc(node):
+    kind = node[0]
+    if kind == "num":
+        return node[1]
+    if kind == "neg":
+        return -_evaluate_calc(node[1])
+    _, op, left_node, right_node = node
+    left = _evaluate_calc(left_node)
+    right = _evaluate_calc(right_node)
+    if op == "+":
+        return left + right
+    if op == "-":
+        return left - right
+    if op == "*":
+        return left * right
+    return left / right
+

+def handle_calc(params: dict) -> tuple[int, dict]:

+    if not isinstance(expr, str) or len(expr) > 100:
